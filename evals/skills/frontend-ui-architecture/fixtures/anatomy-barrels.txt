export function lineLabel(f: { start_line: number; end_line: number }): string {
  return f.start_line === f.end_line ? `${f.start_line}` : `${f.start_line}-${f.end_line}`;
}

import type { FindingRecord } from "@devdigest/shared";
import { lineLabel } from "./index";
import { truncate } from "@/lib/utils";
import { s } from "./styles";

export function FindingCard({ finding }: { finding: FindingRecord }) {
  return (
    <div style={s.card}>
      <span style={s.loc}>{finding.file}:{lineLabel(finding)}</span>
      <p>{truncate(finding.body, 240)}</p>
    </div>
  );
}
